import re


def tokenize(input_string):
    reconhecidos = []
    linha = 1

    mo = re.finditer(r'(?P<COMMENT>\#[^\n]*)|'r'(?P<STRING>"[^"\n]*")|'r'(?P<VAR>\?[a-zA-Z0-9]+)|'r'(?P<LANG>\@[a-zA-Z]{2})|'r'(?P<NUM>\d+)|'r'(?P<SELECT>select\b)|'r'(?P<LIMIT>LIMIT\b)|'r'(?P<DELIM>[{}\.])|'r'(?P<ID>[a-zA-Z0-9_:]+)|'r'(?P<SKIP>[ \t\r]+)|'r'(?P<NEWLINE>\n)|'r'(?P<ERRO>.)',input_string,re.IGNORECASE)

    for m in mo:
        dic = m.groupdict()

        if dic['COMMENT']:
            continue

        elif dic['STRING']:
            t = ("STRING", dic['STRING'], linha, m.span())

        elif dic['VAR']:
            t = ("VAR", dic['VAR'], linha, m.span())

        elif dic['LANG']:
            t = ("LANG", dic['LANG'], linha, m.span())

        elif dic['NUM']:
            t = ("NUM", dic['NUM'], linha, m.span())

        elif dic['SELECT']:
            t = ("SELECT", dic['SELECT'], linha, m.span())

        elif dic['LIMIT']:
            t = ("LIMIT", dic['LIMIT'], linha, m.span())

        elif dic['DELIM']:
            t = ("DELIM", dic['DELIM'], linha, m.span())

        elif dic['ID']:
            t = ("ID", dic['ID'], linha, m.span())

        elif dic['SKIP']:
            continue

        elif dic['NEWLINE']:
            t = ("NEWLINE", dic['NEWLINE'], linha, m.span())
            reconhecidos.append(t)
            linha += 1
            continue

        elif dic['ERRO']:
            t = ("ERRO", dic['ERRO'], linha, m.span())

        else:
            continue

        reconhecidos.append(t)

    return reconhecidos
